% Les marqueurs du rapport, et le seul endroit du dépôt où une valeur de page de
% garde est écrite. Tout autre fichier les emploie par leur commande, jamais en
% écrivant la valeur.
%
% LES DEUX NOMS DE PERSONNE NE SONT PAS ICI. Ils ne sont commis nulle part : les
% deux marqueurs nominatifs sont déclarés VIDES dans ce fichier, et redéfinis à la
% compilation par `noms.tex`, qui n'est pas suivi par le gestionnaire de versions.
% Voir la fin de ce fichier.
%
% Un marqueur nominatif renseigné ici ferait échouer un contrôle en état de
% brouillon, et `tests/test_aucun_nom_de_personne.py` le verrait aussi.
% `tests/test_marqueurs_nominatifs.py` vérifie l'accord sur CE fichier, jamais sur
% le PDF produit.
%
% Ne pas renommer les commandes : les contrôles les cherchent par leur nom.

% --- l'état du document -------------------------------------------------------
% Deux valeurs, et deux seulement.
%
%   brouillon : le rapport n'est pas remis. Les DEUX marqueurs nominatifs
%               ci-dessous sont TOUS DEUX vides, et la page de garde porte une
%               mention visible de version de travail.
%   remise    : le rapport est remis. Les deux marqueurs sont TOUS DEUX
%               renseignés, et la mention disparaît.
%
% Un contrôle vérifie l'accord entre cet état et les deux marqueurs : renseigner
% l'un des deux est rouge dans les deux états, et c'est l'oubli qui arrive.
% Passer à la remise ne coûte que ce mot — et le contrôle dira alors quel marqueur
% manque encore. Les noms venant de `noms.tex`, la bascule suppose que ce fichier
% existe, ou que l'intégration continue l'écrive depuis ses variables de dépôt.
\newcommand{\etatDuDocument}{brouillon}

% --- marqueurs nominatifs : ils portent un nom de personne ---------------------
% DEUX, et deux seulement. Le document est un rapport de stage d'application : il
% n'a pas d'encadrant académique, pas de président de jury, pas d'examinateur.
% Ces trois marqueurs ont existé et ont été RETIRÉS, non laissés vides — un
% marqueur vide est un oubli qui attend, et une ligne vide sur une page de garde
% se remarque là où une ligne absente ne se remarque pas.
\newcommand{\marqueurAuteur}{}
\newcommand{\marqueurEncadrantProfessionnel}{}

% --- marqueurs de contexte : ils ne portent aucun nom de personne --------------
% Le contrôle des marqueurs nominatifs ne les regarde pas. Ils sont ici pour que
% la page de garde n'écrive aucune valeur en dur.
%
% La date de soutenance n'est pas connue, et reste vide : la ligne qui la porte
% disparaît entièrement plutôt que de composer « Soutenu le » suivi de rien.
\newcommand{\marqueurEtablissementFormation}{Institut National de Statistique et d'Économie Appliquée (INSEA), Rabat}
\newcommand{\marqueurFiliere}{Data \& Software Engineering}
\newcommand{\marqueurAnneeUniversitaire}{2025-2026}
\newcommand{\marqueurOrganismeAccueil}{Hôpital Sidi Saïd, Meknès}
\newcommand{\marqueurDateSoutenance}{}

% --- ce que l'état fait porter au document ------------------------------------
% Le mécanisme est typographique : le document se suffit, et ne dépend d'aucun
% contrôle pour dire ce qu'il est. `\siBrouillon{...}` compose son argument en
% état de brouillon, et rien en état de remise.
%
% La comparaison passe par \ifx sur deux commandes développées : c'est la forme
% que le noyau offre sans paquet supplémentaire, et elle compare les définitions
% caractère pour caractère.
\newcommand{\etatBrouillonAttendu}{brouillon}
\newcommand{\siBrouillon}[1]{%
  \ifx\etatDuDocument\etatBrouillonAttendu#1\fi%
}

% La mention elle-même, écrite une fois et employée par les deux documents.
\newcommand{\mentionVersionDeTravail}{%
  \siBrouillon{\textbf{Version de travail} — document non remis}%
}

% --- aucun texte fixe autour d'un marqueur qui peut être vide ------------------
% `\siRenseigne{\marqueur}{ce qui compose}` ne compose son second argument que si
% le marqueur porte une valeur. Une étiquette — « Soutenu le », « Réalisé par » —
% disparaît donc avec sa valeur, au lieu d'annoncer une phrase qui s'arrête.
%
% LE TÉMOIN DE VACUITÉ EST DÉCLARÉ ICI, ET NON EMPRUNTÉ AU NOYAU. `\ifx\x\empty`
% serait FAUX même sur un marqueur vide : `\newcommand` définit ses commandes avec
% le préfixe `\long`, que `\empty` du noyau ne porte pas, et `\ifx` compare aussi
% les préfixes. Le témoin ci-dessous est défini par `\newcommand`, donc avec le
% même préfixe que les marqueurs — c'est le même motif que `\etatBrouillonAttendu`
% ci-dessus, et il est mesuré par compilation, non supposé.
\newcommand{\marqueurVide}{}
\newcommand{\siRenseigne}[2]{%
  \ifx#1\marqueurVide\else#2\fi%
}

% --- les noms, injectés à la compilation --------------------------------------
% `noms.tex` redéfinit les deux marqueurs nominatifs. Il n'est JAMAIS commis :
% `.gitignore` porte `report/noms.tex`, et un contrôle cherche les deux noms dans
% l'ensemble des fichiers suivis.
%
% DEUX CHEMINS SONT ESSAYÉS, et c'est mesuré : le rapport se compose depuis
% `report/`, où le fichier est `noms.tex` ; la présentation se compose depuis
% `slides/` et charge ce fichier-ci par `\input{../report/marqueurs}`, où le même
% fichier est `../report/noms.tex`. Un seul des deux chemins existe à la fois.
% Si aucun ne répond, les marqueurs restent vides et le document compose sans
% noms : c'est le comportement attendu tant que l'état vaut `brouillon`.
\IfFileExists{noms.tex}{%
  \input{noms.tex}%
}{%
  \IfFileExists{../report/noms.tex}{\input{../report/noms.tex}}{}%
}
