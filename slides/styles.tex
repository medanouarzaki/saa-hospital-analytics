% Les styles de la présentation.
%
% Ils sont écrits ici plutôt qu'empruntés à `report/images.tex` : ce fichier-là porte l'apparat des
% figures du rapport — emplacements réservés, logotypes, compteur de captures — dont une planche
% n'a que faire.
%
% UNE SEULE COULEUR D'ACCENT, celle de la page de garde du rapport, à l'identique. Fond blanc, aucun
% dégradé, aucune ombre, aucune barre de navigation : le seul apparat est le numéro de planche en
% pied de page. Le gris ne porte que les mentions secondaires.

\definecolor{accentgarde}{RGB}{23,54,93}
\definecolor{griscartouche}{gray}{0.42}

\setbeamercolor{background canvas}{bg=white}
\setbeamercolor{normal text}{fg=black}
\setbeamercolor{structure}{fg=accentgarde}
\setbeamercolor{frametitle}{fg=accentgarde,bg=}
\setbeamercolor{title}{fg=accentgarde}
\setbeamercolor{page number in head/foot}{fg=griscartouche}
\setbeamerfont{frametitle}{series=\bfseries,size=\Large}
\setbeamerfont{page number in head/foot}{size=\footnotesize}
\setbeamertemplate{navigation symbols}{}
\setbeamertemplate{footline}{%
  \hfill\usebeamercolor[fg]{page number in head/foot}%
  \usebeamerfont{page number in head/foot}\insertframenumber\hspace{6mm}\vspace{3mm}}
\setbeamertemplate{itemize item}{\color{accentgarde}\textbullet}
\setbeamertemplate{itemize subitem}{\color{griscartouche}--}
\setbeamertemplate{enumerate item}{\color{accentgarde}\bfseries\insertenumlabel.}
\setbeamersize{text margin left=10mm, text margin right=10mm}

\usetikzlibrary{positioning, calc, arrows.meta, shapes.geometric}

\tikzset{
  % — une brique de couche, ou une étape.
  briquePres/.style={
    rectangle, draw=accentgarde, line width=0.7pt, align=center,
    font=\small, inner sep=3pt, minimum height=9mm, text width=20mm,
  },
  % — un cadre secondaire : trait pointillé.
  briqueOptPres/.style={briquePres, dash pattern=on 1.6pt off 1.6pt},
  % — la brique d'arrivée : pleine, en couleur d'accent.
  briqueFinPres/.style={briquePres, fill=accentgarde, text=white},
  % — une flèche de flux.
  fluxPres/.style={draw=accentgarde, line width=0.7pt, -{Latex[length=2mm,width=1.6mm]}},
  % — une note grise.
  notePres/.style={font=\footnotesize\itshape, text=griscartouche, align=center},
}

% Un chiffre mis en avant : grand, gras, en couleur d'accent.
\newcommand{\grandChiffre}[1]{{\LARGE\bfseries\color{accentgarde}#1}}
% Une mention secondaire.
\newcommand{\secondaire}[1]{{\color{griscartouche}#1}}
